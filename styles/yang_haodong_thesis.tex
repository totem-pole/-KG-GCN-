% 杨昊东学位论文可见版式的LaTeX统一样式层
% 参数依据 /mnt/data/杨昊东大论文.docx 的OOXML/Word样式直接核对：A4、页边距、字号、行距、标题层级、题注、三线表及参考文献。
% 学校封面、作者、导师等个人元数据因当前未提供，不在此文件中编造。

\usepackage{amsmath,amssymb,graphicx,booktabs,multirow}
\usepackage{geometry}
\usepackage{caption}
\usepackage{fancyhdr}
\usepackage{chngcntr}
\usepackage{setspace}
\usepackage{indentfirst}
\usepackage{titlesec}

\usepackage{hyperref}
\usepackage{etoolbox}

% ---------- 页面 ----------
% 杨昊东原稿：A4；上/下约2.0 cm，左/右约2.5 cm；主要正文节footer distance约1.75 cm。
\geometry{a4paper,top=2.00cm,bottom=2.00cm,left=2.50cm,right=2.50cm,headheight=14pt,headsep=0.50cm,footskip=0.25cm}

% ---------- 字体与正文 ----------
% 原稿中文正文为宋体，英文/数字为Times New Roman。TeX环境采用TeX Gyre Termes作为Times New Roman的开源等距替代。
\setmainfont{TeX Gyre Termes}
\songti\zihao{-4}
% 原稿“论文正文”：12 pt、1.25倍行距、首行约24 pt（2字符）。
\setstretch{1.25}
\setlength{\parindent}{2em}
\setlength{\parskip}{0pt}

% ---------- 章/节标题 ----------
% 原稿：章16 pt黑体加粗居中；1.1为14 pt宋体加粗；1.1.1为12 pt黑体。
\ctexset{
  section = {
    name = {第,章},
    number = \chinese{section},
    format = \centering\heiti\bfseries\zihao{3},
    beforeskip = 16.3pt,
    afterskip = 16.3pt
  },
  subsection = {
    format = \songti\bfseries\zihao{4},
    beforeskip = 8.15pt,
    afterskip = 8.15pt
  },
  subsubsection = {
    format = \heiti\mdseries\zihao{-4},
    beforeskip = 8.15pt,
    afterskip = 8.15pt
  }
}

% 每章另起一页。
\pretocmd{\section}{\clearpage}{}{}

\renewcommand{\thesubsection}{\arabic{section}.\arabic{subsection}}
\renewcommand{\thesubsubsection}{\arabic{section}.\arabic{subsection}.\arabic{subsubsection}}

% ---------- 图、表、公式 ----------
\counterwithin{figure}{section}
\counterwithin{table}{section}
\numberwithin{equation}{section}
\renewcommand{\thefigure}{\arabic{section}-\arabic{figure}}
\renewcommand{\thetable}{\arabic{section}-\arabic{table}}
\renewcommand{\theequation}{\arabic{section}-\arabic{equation}}
\renewcommand{\figurename}{图}
\renewcommand{\tablename}{表}

% 原稿图例：宋体/Times New Roman 10.5 pt、单倍行距、居中。
\DeclareCaptionFont{yangcaption}{\songti\zihao{5}\setstretch{1.0}}
\captionsetup{font=yangcaption,labelsep=space,justification=centering,singlelinecheck=true}
\captionsetup[table]{position=top,skip=5pt}
\captionsetup[figure]{position=bottom,skip=5pt}

% 原稿表格：10.5 pt、单倍行距、首行加粗；上下/表头分隔线约0.5 pt。
\setlength{\heavyrulewidth}{0.5pt}
\setlength{\lightrulewidth}{0.5pt}
\setlength{\cmidrulewidth}{0.5pt}
\AtBeginEnvironment{tabular}{\songti\zihao{5}\setstretch{1.0}}
\let\YangOriginalToprule\toprule
\let\YangOriginalMidrule\midrule
\renewcommand{\toprule}{\bfseries\YangOriginalToprule}
\renewcommand{\midrule}{\mdseries\YangOriginalMidrule}

% ---------- 页眉页脚 ----------
\pagestyle{fancy}
\fancyhf{}
\fancyfoot[C]{\songti\zihao{5}\thepage}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}

% ---------- 目录 ----------
% 原稿TOC一级条目12 pt加粗，二/三级12 pt常规；目录列至三级。
\setcounter{tocdepth}{3}
\renewcommand{\contentsname}{目录}
\renewcommand{\cftsecfont}{\songti\zihao{-4}\bfseries}
\renewcommand{\cftsubsecfont}{\songti\zihao{-4}\mdseries}
\renewcommand{\cftsubsubsecfont}{\songti\zihao{-4}\mdseries}
\renewcommand{\cftsecpagefont}{\songti\zihao{-4}\bfseries}
\renewcommand{\cftsubsecpagefont}{\songti\zihao{-4}\mdseries}
\renewcommand{\cftsubsubsecpagefont}{\songti\zihao{-4}\mdseries}

% ---------- PDF内部链接 ----------
\hypersetup{hidelinks,unicode=true}

% ---------- 摘要 ----------

\newcommand{\cnabstracttitle}{%
